// Patient Table
Table STG_EHP__PATN { 
  PAT_ID VARCHAR // Unique identifier for the patient
  F_NAME VARCHAR // Patient's first name
  L_NAME VARCHAR // Patient's last name / surname
  M_NAME VARCHAR // Patient's middle name
  GEN_CD INT // Numeric code representing gender
  GEN_DES VARCHAR // Text description of gender
  DOB DATE // Patient's date of birth
  CON_NO VARCHAR // Patient's contact phone number
  EM_ADD VARCHAR // Patient's email address
  EC_F_NAME VARCHAR // Emergency contact first name
  EC_L_NAME VARCHAR // Emergency contact last name
  EC_M_NAME VARCHAR // Emergency contact middle name
  EC_CON_NO VARCHAR // Emergency contact phone number
  B_TYPE VARCHAR // Patient's blood type (e.g., A+, O-)
  MAR_ST VARCHAR // Marital status (e.g., Single, Married)
  NAT_ID_S VARCHAR // State that issued the national ID
  NAT_ID_T VARCHAR // Type of national ID (e.g., Passport, SSN)
  NAT_ID_N VARCHAR // National ID number
  NAT_ID_E DATE // Expiry date of the national ID
}
